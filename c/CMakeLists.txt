cmake_minimum_required(VERSION 3.16)
project(dito_native VERSION 0.3.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-c)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3)
pkg_check_modules(CURSES REQUIRED IMPORTED_TARGET ncursesw)
pkg_check_modules(XML REQUIRED IMPORTED_TARGET libxml-2.0)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(dito_core STATIC src/common.c src/config.c src/stream.c src/model.c src/session.c src/storage.c src/tools.c src/agent.c src/render.c src/mcp.c src/web.c src/compact.c src/tui.c src/workbench.c src/legacy.c)
target_include_directories(dito_core PUBLIC include)
target_compile_definitions(dito_core PUBLIC _GNU_SOURCE _POSIX_C_SOURCE=200809L)
target_compile_definitions(dito_core PUBLIC DITO_VERSION="0.3.0-alpha")
target_compile_options(dito_core PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wformat=2)
target_link_libraries(dito_core PUBLIC PkgConfig::CURL PkgConfig::JSON PkgConfig::SQLITE PkgConfig::CURSES PkgConfig::XML Threads::Threads OpenSSL::Crypto m)
add_executable(dito src/main.c src/server.c src/channels.c)
target_link_libraries(dito PRIVATE dito_core)
target_compile_options(dito PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wformat=2)
option(DITO_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(DITO_SANITIZE)
  target_compile_options(dito_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(dito_core PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE dito_core)
target_compile_options(test_core PRIVATE -UNDEBUG)
add_test(NAME core COMMAND test_core)
add_executable(test_workbench tests/test_workbench.c)
target_link_libraries(test_workbench PRIVATE dito_core)
target_compile_options(test_workbench PRIVATE -UNDEBUG)
add_test(NAME workbench COMMAND test_workbench)
install(TARGETS dito RUNTIME DESTINATION lib/dito)
install(DIRECTORY ../config ../personas ../identities ../system-prompts ../kb ../skills ../web-ui DESTINATION lib/dito)
install(FILES ../LICENSE ../NOTICE DESTINATION lib/dito)
install(FILES third_party/MiyuAgent-LICENSE DESTINATION lib/dito/third_party)
install(CODE "file(MAKE_DIRECTORY \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/bin\")\nfile(CREATE_LINK \"../lib/dito/dito\" \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/bin/dito\" SYMBOLIC)")
